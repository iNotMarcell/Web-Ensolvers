#!/bin/bash

# Script de inicio automatizado para Sistema de Notas
# Compatible con Linux y macOS

set -e  # Salir si hay algún error

# Colores para output
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m' # No Color

# Banner
echo -e "${BLUE}"
echo "╔═══════════════════════════════════════╗"
echo "║    Sistema de Notas - Iniciador   ║"
echo "╔═══════════════════════════════════════╗"
echo -e "${NC}"

# Función para mostrar mensajes
print_success() {
    echo -e "${GREEN} $1${NC}"
}

print_error() {
    echo -e "${RED} $1${NC}"
}

print_info() {
    echo -e "${BLUE}  $1${NC}"
}

print_warning() {
    echo -e "${YELLOW}  $1${NC}"
}

# Verificar Node.js
print_info "Verificando Node.js..."
if ! command -v node &> /dev/null; then
    print_error "Node.js no está instalado"
    echo "Por favor instala Node.js 18.17.0 o superior desde https://nodejs.org/"
    exit 1
fi

NODE_VERSION=$(node --version | cut -d'v' -f2 | cut -d'.' -f1)
if [ "$NODE_VERSION" -lt 18 ]; then
    print_error "Node.js versión $NODE_VERSION detectada. Se requiere versión 18 o superior"
    exit 1
fi
print_success "Node.js $(node --version) detectado"

# Verificar npm
print_info "Verificando npm..."
if ! command -v npm &> /dev/null; then
    print_error "npm no está instalado"
    exit 1
fi
print_success "npm $(npm --version) detectado"

# Verificar archivo .env
print_info "Verificando configuración..."
if [ ! -f "backend/.env" ]; then
    print_warning "Archivo .env no encontrado"
    
    if [ -f "backend/.env.example" ]; then
        print_info "Copiando .env.example a .env"
        cp backend/.env.example backend/.env
        print_warning "Por favor edita backend/.env con tus credenciales de SQL Server"
        print_info "Presiona Enter después de configurar .env..."
        read
    else
        print_info "Creando archivo .env con valores por defecto..."
        cat > backend/.env << EOF
DB_HOST=localhost
DB_PORT=1433
DB_USER=sa
DB_PASS=mysql
DB_NAME=notes_app
PORT=3000
EOF
        print_warning "Archivo .env creado. Verifica las credenciales en backend/.env"
        print_info "Presiona Enter para continuar..."
        read
    fi
fi
print_success "Configuración verificada"

# Instalar dependencias del backend
print_info "Instalando dependencias del backend..."
cd backend
if [ ! -d "node_modules" ]; then
    npm install
    print_success "Dependencias del backend instaladas"
else
    print_info "Dependencias del backend ya instaladas"
fi
cd ..

# Instalar dependencias del frontend
print_info "Instalando dependencias del frontend..."
cd frontend
if [ ! -d "node_modules" ]; then
    npm install
    print_success "Dependencias del frontend instaladas"
else
    print_info "Dependencias del frontend ya instaladas"
fi
cd ..

# Verificar conexión a SQL Server
print_info "Verificando conexión a SQL Server..."
print_warning "Asegúrate de que SQL Server esté corriendo en localhost:1433"

# Crear directorio para logs
mkdir -p logs

# Función para limpiar procesos al salir
cleanup() {
    print_info "\n Deteniendo servidores..."
    kill $BACKEND_PID $FRONTEND_PID 2>/dev/null
    print_success "Servidores detenidos"
    exit 0
}

trap cleanup SIGINT SIGTERM

# Iniciar backend
print_info "Iniciando backend en puerto 3000..."
cd backend
npm start > ../logs/backend.log 2>&1 &
BACKEND_PID=$!
cd ..
sleep 3

# Verificar que el backend inició correctamente
if ps -p $BACKEND_PID > /dev/null; then
    print_success "Backend iniciado correctamente (PID: $BACKEND_PID)"
else
    print_error "Error al iniciar el backend"
    cat logs/backend.log
    exit 1
fi

# Iniciar frontend
print_info "Iniciando frontend en puerto 3001..."
cd frontend
BROWSER=none npm start > ../logs/frontend.log 2>&1 &
FRONTEND_PID=$!
cd ..
sleep 5

# Verificar que el frontend inició correctamente
if ps -p $FRONTEND_PID > /dev/null; then
    print_success "Frontend iniciado correctamente (PID: $FRONTEND_PID)"
else
    print_error "Error al iniciar el frontend"
    cat logs/frontend.log
    kill $BACKEND_PID
    exit 1
fi

# Mostrar información de acceso
echo ""
echo -e "${GREEN}╔═══════════════════════════════════════════════════╗${NC}"
echo -e "${GREEN}║   ¡Aplicación iniciada exitosamente!          ║${NC}"
echo -e "${GREEN}╚═══════════════════════════════════════════════════╝${NC}"
echo ""
print_info "📱 Frontend: http://localhost:3001"
print_info "🔌 Backend:  http://localhost:3000"
print_info "📊 API Docs: http://localhost:3000/api"
echo ""
print_info "🔐 Credenciales por defecto:"
echo "   Usuario: admin"
echo "   Contraseña: admin123"
echo ""
print_warning "Presiona Ctrl+C para detener los servidores"
echo ""
print_info "📝 Logs disponibles en:"
echo "   Backend:  logs/backend.log"
echo "   Frontend: logs/frontend.log"
echo ""

# Mantener el script corriendo y mostrar logs
tail -f logs/backend.log logs/frontend.log
